cmake_minimum_required(VERSION 3.21)
project(FormaBox VERSION 0.1.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
    # The Windows loader looks beside FormaBox.exe for the deployed Qt DLLs.
    set(CMAKE_INSTALL_BINDIR ".")
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2)
qt_standard_project_setup(REQUIRES 6.5)
qt_add_executable(FormaBox WIN32 desktop/main.cpp desktop/controller.cpp desktop/controller.h)
set_source_files_properties(desktop/qml/Main.qml PROPERTIES QT_RESOURCE_ALIAS Main.qml)
qt_add_qml_module(FormaBox URI FormaBox VERSION 1.0 QML_FILES desktop/qml/Main.qml)
target_link_libraries(FormaBox PRIVATE Qt6::Quick Qt6::QuickControls2)
target_compile_definitions(FormaBox PRIVATE FORMABOX_VERSION="${PROJECT_VERSION}")
if(MSVC)
    target_compile_options(FormaBox PRIVATE /utf-8)
endif()
install(TARGETS FormaBox RUNTIME DESTINATION . BUNDLE DESTINATION .)
install(DIRECTORY worker/formabox DESTINATION worker PATTERN "__pycache__" EXCLUDE)
if(WIN32 AND Qt6_VERSION VERSION_GREATER_EQUAL 6.8)
    set(deploy_options DEPLOY_TOOL_OPTIONS --no-opengl-sw --no-system-d3d-compiler
        --no-system-dxc-compiler --no-compiler-runtime)
endif()
qt_generate_deploy_qml_app_script(TARGET FormaBox OUTPUT_SCRIPT deploy_script
    NO_UNSUPPORTED_PLATFORM_ERROR ${deploy_options})
install(SCRIPT ${deploy_script})
