cmake_minimum_required(VERSION 3.21)
project(FormaBox VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
    enable_language(RC)
    # The Windows loader looks beside FormaBox.exe for the deployed Qt DLLs.
    set(CMAKE_INSTALL_BINDIR ".")
    add_executable(FormaBoxLauncher WIN32 desktop/launcher_win.cpp)
    set_target_properties(FormaBoxLauncher PROPERTIES
        OUTPUT_NAME FormaBox
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/launcher/$<CONFIG>")
    target_link_libraries(FormaBoxLauncher PRIVATE user32)
    if(MSVC)
        target_compile_options(FormaBoxLauncher PRIVATE /utf-8)
        set_property(TARGET FormaBoxLauncher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    elseif(MINGW)
        target_link_options(FormaBoxLauncher PRIVATE -municode -static)
    endif()
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2)
qt_standard_project_setup(REQUIRES 6.5)
qt_add_executable(FormaBox WIN32 desktop/main.cpp desktop/controller.cpp desktop/controller.h)
set(branding_dir "${CMAKE_CURRENT_SOURCE_DIR}/desktop/assets")
qt_add_resources(FormaBox "branding"
    PREFIX "/branding"
    BASE "${branding_dir}"
    FILES "${branding_dir}/formabox.png" "${branding_dir}/button-light.png")
if(WIN32)
    set(icon_resource "${branding_dir}/formabox.rc")
    set_source_files_properties("${icon_resource}" PROPERTIES
        OBJECT_DEPENDS "${branding_dir}/formabox.ico")
    foreach(target FormaBox FormaBoxLauncher)
        target_sources(${target} PRIVATE "${icon_resource}")
        target_include_directories(${target} PRIVATE "$<$<COMPILE_LANGUAGE:RC>:${branding_dir}>")
    endforeach()
endif()
set(formabox_qml Main.qml PdfWorkbench.qml EngineSettings.qml TaskResults.qml RenamePreview.qml)
foreach(qml IN LISTS formabox_qml)
    set_source_files_properties("desktop/qml/${qml}" PROPERTIES QT_RESOURCE_ALIAS "${qml}")
endforeach()
list(TRANSFORM formabox_qml PREPEND "desktop/qml/")
qt_add_qml_module(FormaBox URI FormaBox VERSION 1.0 QML_FILES ${formabox_qml})
target_link_libraries(FormaBox PRIVATE Qt6::Quick Qt6::QuickControls2)
target_compile_definitions(FormaBox PRIVATE FORMABOX_VERSION="${PROJECT_VERSION}")
if(MSVC)
    target_compile_options(FormaBox PRIVATE /utf-8)
endif()
install(TARGETS FormaBox RUNTIME DESTINATION . BUNDLE DESTINATION .)
install(DIRECTORY worker/formabox DESTINATION worker PATTERN "__pycache__" EXCLUDE)
if(WIN32 AND Qt6_VERSION VERSION_GREATER_EQUAL 6.8)
    set(deploy_options DEPLOY_TOOL_OPTIONS --no-opengl-sw --no-system-d3d-compiler
        --no-system-dxc-compiler --no-compiler-runtime)
endif()
qt_generate_deploy_qml_app_script(TARGET FormaBox OUTPUT_SCRIPT deploy_script
    NO_UNSUPPORTED_PLATFORM_ERROR ${deploy_options})
install(SCRIPT ${deploy_script})
